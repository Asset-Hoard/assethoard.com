Linux package managers

Prefer to install from a repository so you get automatic updates? Add the AssetHoard repo for your distribution and install from the command line.

curl -fsSL https://releases.assethoard.com/apt/assethoard-archive-keyring.gpg \
  | sudo gpg --dearmor -o /usr/share/keyrings/assethoard-archive-keyring.gpg

echo "deb [arch=$(dpkg --print-architecture) signed-by=/usr/share/keyrings/assethoard-archive-keyring.gpg] https://releases.assethoard.com/apt stable main" \
  | sudo tee /etc/apt/sources.list.d/assethoard.list

sudo apt update && sudo apt install asset-hoard
